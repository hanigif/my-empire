from flask import Flask, render_template
import plotly.graph_objs as go
import pandas as pd
import os

app = Flask(__name__)

@app.route('/')
def index():
    # بيانات وهمية للـ 10 شركات (سنحدثها لاحقاً لبيانات حية)
    data = {'Company': ['Apple', 'Nvidia', 'Google', 'Microsoft', 'Tesla'],
            'Profit_Prob': [85, 92, 78, 88, 70]}
    df = pd.DataFrame(data)
    
    fig = go.Figure(data=[go.Bar(x=df['Company'], y=df['Profit_Prob'], marker_color='gold')])
    fig.update_layout(title='توقعات الربح للأسبوع القادم %', template='plotly_dark')
    
    chart_html = fig.to_html(full_html=False)
    return render_template('index.html', chart=chart_html)

if __name__ == "__main__":
    port = int(os.environ.get("PORT", 5000))
    app.run(host='0.0.0.0', port=port)
